## 问题现象

客户商品编码通过分类自动增长后编码太大，导致无法通过分类树新增商品编码。

## 问题原因

客户软件使用中操作错误，导致编码太大，商品编号无法进行自动增长！

## 解决方案

通过SQL语句重新更新商品编号。
--分类更新商品
select distinct spfl into #tmp from dbo.spkfk\_dr
select spfl,ROW\_NUMBER() OVER (ORDER BY getdate() asc )as age into #tmp2 from #tmp
select spbh,spid into #tmp3 from spkfk where 1\&lt;\&gt;1
declare @i int,
@max int
set @i=1
select @max=max(age) from #tmp2 
while @i\&lt;=@max
begin 
insert into #tmp3 ( spbh, spid )
select rtrim((select spfl from #tmp2 where age=@i))
\+REPLICATE('0',7-len(rtrim((select spfl from #tmp2 where age=@i)))-len(convert(char(10),ROW\_NUMBER() OVER (ORDER BY getdate() asc))))\+convert(char(10),ROW\_NUMBER() OVER (ORDER BY getdate() asc))
,spid 
--update spkfk\_dr set spbh=rtrim((select spfl from #tmp2 where age=@i))
--\+REPLICATE('0',7-len(rtrim((select spfl from #tmp2 where age=@i)))-len(convert(char(10),ROW\_NUMBER() OVER (ORDER BY getdate() asc))))\+convert(char(10),ROW\_NUMBER() OVER (ORDER BY getdate() asc))
from dbo.spkfk\_dr where spfl in (select spfl from #tmp2 where age=@i)
set @i=@i\+1
end
update b set spbh=a.spbh
 from #tmp3 a,dbo.spkfk\_dr b where a.spid=b.spid 
drop table #tmp
drop table #tmp2
drop table #tmp3
