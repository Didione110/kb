## 解决方案

例子：num DECIMAL(14,2) 修改为num DECIMAL(14,3)
CREATE PROCEDURE dbo.usp_AlterNumColumnTo
AS
BEGIN
 SET NOCOUNT ON;

 -- 创建临时表记录执行日志
 CREATE TABLE #AlterLog (
 TableName NVARCHAR(255),
 Script NVARCHAR(MAX),
 Status NVARCHAR(50),
 ErrorMsg NVARCHAR(MAX),
 ExecTime DATETIME DEFAULT GETDATE()
 );

 -- 定义变量
 DECLARE @TableSchema NVARCHAR(128);
 DECLARE @TableName NVARCHAR(128);
 DECLARE @Sql NVARCHAR(MAX);
 DECLARE @ErrorMsg NVARCHAR(MAX);

 -- 游标遍历所有包含 num 字段且类型为 DECIMAL(14,2) 的表
 DECLARE cur CURSOR FOR
 SELECT 
 TABLE_SCHEMA, 
 TABLE_NAME
 FROM INFORMATION_SCHEMA.COLUMNS
 WHERE 
 COLUMN_NAME = 'num '
 AND DATA_TYPE = 'decimal'
 AND NUMERIC_PRECISION = 14
 AND NUMERIC_SCALE = 2;

 OPEN cur;
 FETCH NEXT FROM cur INTO @TableSchema, @TableName;

 WHILE @@FETCH_STATUS = 0
 BEGIN
 BEGIN TRY
 -- 生成 ALTER 语句
 SET @Sql = N'
 ALTER TABLE ' + QUOTENAME(@TableSchema) + '.' + QUOTENAME(@TableName) + '
 ALTER COLUMN num DECIMAL(14,3);';

 -- 执行 SQL
 EXEC sp_executesql @Sql;

 -- 记录成功日志
 INSERT INTO #AlterLog (TableName, Script, Status)
 VALUES (
 QUOTENAME(@TableSchema) + '.' + QUOTENAME(@TableName),
 @Sql,
 'Success'
 );
 END TRY
 BEGIN CATCH
 -- 捕获错误信息
 SET @ErrorMsg = ERROR_MESSAGE();

 -- 记录失败日志
 INSERT INTO #AlterLog (TableName, Script, Status, ErrorMsg)
 VALUES (
 QUOTENAME(@TableSchema) + '.' + QUOTENAME(@TableName),
 @Sql,
 'Failed',
 @ErrorMsg
 );
 END CATCH

 FETCH NEXT FROM cur INTO @TableSchema, @TableName;
 END

 CLOSE cur;
 DEALLOCATE cur;

 -- 返回执行结果
 SELECT * FROM #AlterLog;
END
