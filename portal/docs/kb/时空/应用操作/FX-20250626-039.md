## 问题现象

DHERP在Oracle环境下，标准默认查询均采用SQL宏的形式来加载语句展现
涉及到比较复杂的关联查询报表，无法用单一的SQL语句进行展现，需要配合存储过程T-SQL来进行处理

## 问题原因

标准模板内无对应方法

## 解决方案

调用方式参照：
call k_view_szyxbtc(c_cursor=&gt;:__oracle_cursor,p_StartDate =&gt;'2025-01-01', p_EndDate =&gt;'2025-01-31', p_ViewType=&gt;'1')

对应存储过程的案例参照：
create or replace procedure k_view_szyxbtc
 ( c_cursor out sys_refcursor,
 p_StartDate IN VARCHAR2, --开始时间
 p_EndDate IN VARCHAR2, --结束时间
 p_EntId IN VARCHAR2,--企业ID
 p_ViewType IN VARCHAR2 --报表类型

 
 )
AS
begin
 
 -- 数字营销部外部采购非集团产品提成
 IF p_ViewType = '1' THEN
 declare v_ml decimal(18,2):=0; 
 v_fy_yunfei decimal(18,2):=0; 
 v_ml_jtzb decimal(18,2):=0; 
 v_ml_wb_zjcg decimal(18,2):=0; 
 v_ml_wb_zssj decimal(18,2):=0; 
 v_ml_wb_cgtd decimal(18,2):=0; 
 v_ml_jlr_hz decimal(18,2):=0; 
 v_ml_jlr_cgb decimal(18,2):=0; 
 v_ml_jlr_zgb decimal(18,2):=0; 
 v_ml_jlr_ccb decimal(18,2):=0; 
 v_ml_jlr_ddfh decimal(18,2):=0; 
 v_ml_jlr_htgl decimal(18,2):=0; 
 v_ml_jlr_szyxxj decimal(18,2):=0; 
 v_ml_jlr_xxb decimal(18,2):=0; 
 v_ml_jlr_szyxfzr decimal(18,2):=0; 
 v_ml_jlr_szyxb decimal(18,2):=0; 
 v_ml_jlr_xsry decimal(18,2):=0; 
 v_ml_jlr_xszz decimal(18,2):=0; 
 v_ml_jlr_szyxfzr1 decimal(18,2):=0; 
 v_ml_jlr_gszht decimal(18,2):=0; 
 v_ml_jlr_szyxfzr2 decimal(18,2):=0; 
 v_ml_jlr_szyxgs decimal(18,2):=0; 
 begin
 --计算数字营销部部门总毛利
 select sum( b.num*( b.taxprice - (nvl(e.gebjj,0) + ((b.taxprice - nvl(e.gebjj,0))*0.13)))) into v_ml 
 from saleoutmt a 
 join saleoutdt b on a.billno = b.billno and a.entid = b.entid 
 left join batchcode e on e.goodsid = b.goodsid and e.angleid = b.angleid and e.entid = b.entid 
 where a.dates &gt;= p_StartDate and a.dates &lt;= p_EndDate and a.b2btype like '线上-%' and a.entid = p_EntId;
 --计算平台运营费用|运费
 select sum(a.k_kdcost) into v_fy_yunfei 
 from saleoutmt a 
 where a.dates &gt;= p_StartDate and a.dates &lt;= p_EndDate and a.b2btype like '线上-%' and a.entid = p_EntId; 
 --计算扣掉平台运营费用后剩下的数字营销部部门毛利|集团总部商业公司（20%）
 select (v_ml - v_fy_yunfei)*0.2 into v_ml_jtzb from dual ;
 --扣掉平台运营费用后剩下的数字营销部部门毛利（80%）|外部采购团队（15%）|直接采购人员 （50%）
 select (v_ml - v_fy_yunfei)*0.8*0.15*0.5 into v_ml_wb_zjcg from dual ; 
 --扣掉平台运营费用后剩下的数字营销部部门毛利（80%）|外部采购团队（15%）|采购人员直属上级（20%） 
 select (v_ml - v_fy_yunfei)*0.8*0.15*0.2 into v_ml_wb_zssj from dual ; 
 --扣掉平台运营费用后剩下的数字营销部部门毛利（80%）|外部采购团队（15%）|采购团队剩余部分 （30%） 
 select (v_ml - v_fy_yunfei)*0.8*0.15*0.3 into v_ml_wb_cgtd from dual ; 
 --扣掉平台运营费用后剩下的数字营销部部门毛利（80%）|数字营销部部门净利润 |净利润汇总
 select (v_ml - v_fy_yunfei)*0.8*0.85 into v_ml_jlr_hz from dual ; 
 --扣掉平台运营费用后剩下的数字营销部部门毛利（80%）|数字营销部部门净利润 |采购服务团队（15%）|采购部（2%） 
 select v_ml_jlr_hz*0.15*0.02 into v_ml_jlr_cgb from dual ;
