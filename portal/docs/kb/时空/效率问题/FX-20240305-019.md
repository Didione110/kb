## 问题现象

时空V10、V11中，数据行大于2W行左右不能通过系统自带导出Execl表功能导出数据。

## 问题原因

数据量大无法导出

## 解决方案

新建并调用csv导出函数：
function uf_getExcel()&#123;

 toolbar.enableItem("toolbar.m4",false);
 
 try &#123;
 //sql宏参数
 var params = &#123;&#125;;
 params.StartDate = DsMain.field("StartDate").value;
 params.EndDate = DsMain.field("EndDate").value;
 
 //导出参数
 var csvPars = &#123;&#125;;
 csvPars.openmode = "sql";
 csvPars.openparam = "uf_getExcelmx_sql"; //sql宏的名字
 csvPars.filename = "销售出库明细"+DsMain.field("StartDate").value+"-"+DsMain.field("EndDate").value+".csv"; //导出文件名
 csvPars.isExpCNFildName = "Y"; //是否包含表头
 csvPars.charset = "gb2312"; //"utf-8"
 csvPars.variables = params;
 //运行
 WebService.asynDownload("utility.exportCsv", csvPars); 
 //alert("后台数据准备中，请稍候");
 &#125; finally &#123;
 toolbar.enableItem("toolbar.m4",true);
 &#125;
