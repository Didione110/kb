## 解决方案

增加鼠标悬停功能显示完整数据
function bindTooltip(ctrl, delay = 100) &#123;
 if (!ctrl || !ctrl.dom) return;
 setTimeout(function() &#123;
 var inputDom = ctrl.dom;
 var tooltip = document.getElementById('globalTooltip');
 if (!tooltip) &#123;
 tooltip = document.createElement('div');
 tooltip.id = 'globalTooltip';
 tooltip.style.cssText = `
 position: absolute;
 background-color: #333;
 color: #fff;
 border-radius: 4px;
 padding: 4px 8px;
 font-size: 12px;
 white-space: nowrap;
 z-index: 10000;
 display: none;
 box-shadow: 0 2px 5px rgba(0,0,0,0.2);
 pointer-events: none;
 `;
 document.body.appendChild(tooltip);
 &#125;
 var show = function() &#123;
 var text = ctrl.getvalue ? ctrl.getvalue() : inputDom.value;
 if (!text) text = '暂无内容';
 tooltip.textContent = text;
 tooltip.style.display = 'block';
 var rect = inputDom.getBoundingClientRect();
 var scrollTop = window.pageYOffset || document.documentElement.scrollTop;
 tooltip.style.left = rect.left + 'px';
 tooltip.style.top = (rect.bottom + scrollTop + 5) + 'px';
 &#125;;
 var hide = function() &#123;
 tooltip.style.display = 'none';
 &#125;;
 inputDom.addEventListener('mouseenter', show);
 inputDom.addEventListener('mouseleave', hide);
 &#125;, delay);
&#125;

var textboxesToBind = [textbox3, textbox21]; // 按需添加
for (var i = 0; i &lt; textboxesToBind.length; i++) &#123;
 bindTooltip(textboxesToBind[i]);
&#125;

textbox3, textbox21为所需悬停功能的textbox name
