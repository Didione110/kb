菜单进入使用次数统计

## 解决方案

select b.MenuName,b.CatCode,count(*) as 使用次数
 From syslog a inner join menutree b on a.programid=b.programid
 left join menulist c on c.templetid = b.templetid and c.menutype &lt;&gt; 3 and c.beactive = 'Y'
 where a.createtime &gt;='2024-08-21 00:00:00'
 group by b.MenuName,b.CatCode
