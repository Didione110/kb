## 问题现象

数据库查询语句xml类型转换

## 问题原因

有时三方接口返回表数据结构为XML格式，需要转换为表字段格式，需用sql语句处理转换

## 解决方案

使用cross apply content.nodes()函数，一键打印获取发票结果示例：
select b.billno,b.fphm,b.fpdm
,a.r.value('@InvoiceNumber','varchar(50)') as InvoiceNumber,a.r.value('@CustomerName','varchar(50)') as businessname
,a.r.value('@CustomerTaxCode','varchar(50)') as CustomerTaxCode,a.r.value('@AddressAndPhone','varchar(50)') as AddressAndPhone
,a.r.value('@BankAndAccount','varchar(50)') as BankAndAccount,a.r.value('@SellerBankName','varchar(50)') as SellerBankName
,a.r.value('@KHSJ', 'varchar(50)') as KHSJ,a.r.value('@KHYJ', 'varchar(50)') as KHYJ
,a.r.value('@SellerAdrrPhone','varchar(50)') as SellerAdrrPhone,a.r.value('@SellerTaxCode','varchar(50)') as SellerTaxCode
,a.r.value('@FPZL', 'varchar(50)') as FPZL,a.r.value('@InvoiceMemo', 'varchar(50)') as InvoiceMemo
,a.r.value('@KPR', 'varchar(50)') as KPR,a.r.value('@SKR', 'varchar(50)') as SKR
,a.r.value('@FHR', 'varchar(50)') as FHR,a.r.value('@FPHM', 'varchar(50)') as FPHM
,a.r.value('@CHFPH', 'varchar(50)') as CHFPH,a.r.value('@Amount', 'varchar(50)') as Amount
,a.r.value('@TaxAmount', 'varchar(50)') as TaxAmount,a.r.value('@InTaxAmount', 'varchar(50)') as InTaxAmount
,a.r.value('@TaxRate', 'varchar(50)') as TaxRate,a.r.value('@IsListing', 'varchar(50)') as IsListing
,a.r.value('@JEFS', 'varchar(50)') as JEFS,a.r.value('@RowCount', 'varchar(50)') as RowCount1
,c.r.value('@ProductCode', 'varchar(50)') AS goodscode,c.r.value('@EntryNumber', 'varchar(50)') AS sn
,c.r.value('@ProductName', 'varchar(50)') AS goodsname,c.r.value('@ProductModel', 'varchar(50)') AS goodsspec
,c.r.value('@ProductUnit', 'varchar(50)') AS unit,c.r.value('@TaxRate', 'varchar(50)') AS TaxRate
,c.r.value('@Quantity', 'varchar(50)') AS num,c.r.value('@Price', 'varchar(50)') AS Price
,c.r.value('@HSDJ', 'varchar(50)') AS taxprice,c.r.value('@Amount', 'varchar(50)') AS Amount
,c.r.value('@TaxAmount', 'varchar(50)') AS TaxAmount,c.r.value('@Flbm', 'varchar(50)') AS Flbm
from (SELECT billno,fphm,fpdm,cast(content as xml) as content
FROM dbo.k_prn_invoicelog ) b
cross apply content.nodes('/Config/InvoiceList/InvHead') as a(r)
cross apply content.nodes('/Config/InvoiceList/InvDetail/DetailEntry') as c(r)
WHERE b.fphm = '05801268'
