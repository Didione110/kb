## 问题现象

请购单单据状态由终止修改为核准

## 问题原因

接口联调需要还原单据状态

## 解决方案

declare @prDocNo nvarchar(50)
set @prDocNo='请购单号' --请购单号
declare @orgCode nvarchar(50)='组织code'--组织code
declare @orgID bigint=''--组织id
select @orgID=id from Base_Organization WHERE Code=@orgCode--组织code
select @orgID

update pr set pr.Cancel_CancelDate=null,pr.Cancel_Canceled=0,pr.Cancel_CancelReason=null ,pr.Cancel_CancelUser=''
from PR_PR pr join PR_PRLine prl on pr.ID=prl.pr
where pr.docno=@prDocNo and pr.org=@orgID

update prl set prl.Cancel_CancelDate=null,prl.Cancel_Canceled=0,prl.Cancel_CancelReason=null ,prl.Cancel_CancelUser=''
from PR_PR pr join PR_PRLine prl on pr.ID=prl.pr
where pr.docno=@prDocNo and pr.org=@orgID
