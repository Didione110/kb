## 问题原因

由于业务人员误操作将销售订单终止了，想还原单据状态为正常的状态，已强调用单据“关闭”来处理

## 解决方案

declare @soDocNo nvarchar(50)
set @soDocNo='销售订单号' --销售订单号
declare @orgCode nvarchar(50)='组织code'--组织code
declare @orgID bigint=''--组织id
select @orgID=id from Base_Organization WHERE Code=@orgCode--组织code
select @orgID

update a set a.Cancel_Canceled=0,a.Cancel_CancelDate=NULL,a.Cancel_CancelReason='',a.Cancel_CancelUser=''
from sm_soshipline a 
left join sm_soline b on a.soline=b.id
left join sm_so c on b.so=c.id where c.docno=@soDocNo and c.org=@orgID
update b set b.IsCancel=0 
from sm_soline b 
left join sm_so c on b.so=c.id where c.docno=@soDocNo
update c set c.Cancel_Canceled=0,c.Cancel_CancelDate=NULL,c.Cancel_CancelReason='',c.Cancel_CancelUser=''
from sm_so c where c.docno=@soDocNo and c.org=@orgID

update sm_so set WFCurrentState=3,WFOriginalState=-1 where [Status]=1 and docno=@soDocNo and Org=@orgID
update sm_so set WFCurrentState=0,WFOriginalState=3 where [Status]=2 and docno=@soDocNo and Org=@orgID
update sm_so set WFCurrentState=1,WFOriginalState=0 where [Status]=3 and docno=@soDocNo and Org=@orgID
