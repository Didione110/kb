## 问题现象

【会议管理】数据操作：清理会议的全部测试数据

## 问题原因

部分客户环境安装后测试阶段的数据需要清空

## 解决方案

数据操作请提前做好数据库备份
数据操作务必谨慎，不要完全相信SQL，不同环境存在不同差异，做好充分自测验证
务必确认连接的数据库准确性，千万不要连接错数据库
执行以下脚本在数据库中进行清理，请一定按照步骤执行，并且一定要先备份！！！

-- 1、备份数据表（一定要先备份，非常重要！！！）
create table meeting_bak as select * from meeting
create table MEETING_COMMENT_bak as select * from MEETING_COMMENT
create table MEETING_RESOURCES_bak as select * from MEETING_RESOURCES
create table MEETING_REPLY_bak as select * from MEETING_REPLY
create table MEETING_SUMMARY_bak as select * from MEETING_SUMMARY
create table MEETING_SUMMARY_SCOPE_bak as select * from MEETING_SUMMARY_SCOPE

-- 2、清空对应表数据
delete from meeting
delete from MEETING_COMMENT
delete from MEETING_RESOURCES
delete from MEETING_REPLY
delete from MEETING_SUMMARY
delete from MEETING_SUMMARY_SCOPE
