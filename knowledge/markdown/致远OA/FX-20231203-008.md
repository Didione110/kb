## 解决方案

后台通过REST 接口处理，处理方法如下：
/**
 * 插入附件
 * @param params
 * @return
 * @throws BusinessException
 */
 @POST
 @Produces(MediaType.APPLICATION_JSON)
 @Path("insert")
 public Response insert(Map<String, String> params) throws BusinessException {
 try {
 File attachmentFile = new File("D://test111.pdf");
 FileInputStream fis = new FileInputStream(attachmentFile); 
 //上传附件
 Long fielId = fileuplaodId(fis, attachmentFile); 
 Long[] fileIds = new Long[1];
 fileIds[0] = fielId;
 Long subReference = UUIDLong.longUUID();
 AttachmentManager attachmentManager = (AttachmentManager) AppContext.getBean("attachmentManager");
 attachmentManager.create(fileIds,ApplicationCategoryEnum.cap4Form, ParamUtil.getLong(params, "summaryId"), subReference);
 
 FormApi4Cap4 formApi4Cap4 = (FormApi4Cap4) AppContext.getBean("formApi4Cap4");
 ColManager colManager = (ColManager) AppContext.getBean("colManager");
 ColSummary colSummary = colManager.getSummaryById(ParamUtil.getLong(params, "summaryId"));
 FormBean formBean = formApi4Cap4.getForm(colSummary.getFormAppid());
 FormTableBean masterTableBean = formBean.getMasterTableBean();
 Map<String, Object> dataMap = new HashMap<String, Object>();
 FormFieldBean attachmentField = formBean.getFieldBeanByDisplay("附件1");//表单数据域名称
 dataMap.put(attachmentField.getName(),subReference);
 
 CAP4FormDataDAO cap4FormDataDAO = (CAP4FormDataDAO) AppContext.getBean("cap4FormDataDAO");
 cap4FormDataDAO.updateData(colSummary.getFormRecordid(), masterTableBean.getTableName(), dataMap); 
 } catch (Exception e) {
 LOG.error(e);
 } 
 return ok(new HashMap<String, String>());
 }
