## 问题现象

客户有时需要将表单上的附件信息进行下载，上传至第三方系统进行处理

## 问题原因

需做客开处理

## 解决方案

通过流程超级节点进行REST接口客开处理，通过以下方法获取附件信息
/**
 * 查询附件
 * @param params
 * @return
 * @throws BusinessException
 */
 @POST
 @Produces(MediaType.APPLICATION_JSON)
 @Path("query")
 public Response query(Map<String, String> params) throws BusinessException {
 try {
 AttachmentManager attachmentManager = (AttachmentManager) AppContext.getBean("attachmentManager");
 /**
 * 根据summaryId获取所有附件
 */
 List<Attachment> allAtts = attachmentManager.getAllByReference(ParamUtil.getLong(params, "summaryId"));
 //获取表单内所有附件数据
 List<Attachment> formAttachmentList = new ArrayList<Attachment>();
 for (Attachment att : allAtts) {
 if (att.getCategory().intValue() == ApplicationCategoryEnum.cap4Form.getKey()) {
 formAttachmentList.add(att);
 }
 }
 /**
 * 根据表单masterId获取指定字段附件信息
 */
 FormApi4Cap4 formApi4Cap4 = (FormApi4Cap4) AppContext.getBean("formApi4Cap4");
 ColManager colManager = (ColManager) AppContext.getBean("colManager");
 ColSummary colSummary = colManager.getSummaryById(ParamUtil.getLong(params, "summaryId"));
 FormBean formBean = formApi4Cap4.getForm(colSummary.getFormAppid());
 FormDataMasterBean formDataMasterBean = formApi4Cap4.getDataMasterBeanById(colSummary.getFormRecordid(),formBean, null);
 FormFieldBean attachmentField = formBean.getFieldBeanByDisplay("附件");//表单数据域名称
 Object attachmentFieleValue = formDataMasterBean.getFieldValue(attachmentField.getName());
 List<Long> ids = new ArrayList<Long>();
 ids.add(Long.parseLong(String.valueOf(attachmentFieleValue)));
 List<Attachment> formAttachments = attachmentManager.getBySubReference(ids);
 } catch (SQLException e) {
 
 }
 return ok(new HashMap<String, String>());
 }
