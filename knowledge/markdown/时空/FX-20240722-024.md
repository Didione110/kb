## 问题现象

供应链自带的委托书范围拦截比较蠢，不够人性化，尤其是涉及到商品类型的管控

## 问题原因

智友标准模板使用以商品类型的委托书是采用的contactgoods表进行控制，在dherp里并不适用，改用其他方案来执行

## 解决方案

数据库内新建函数GetWtsStatus

CREATE OR REPLACE Function GetWtsStatus
(

 para_BusinessId In Varchar2,
 para_GoodsId In Varchar2
) Return Varchar2 Is
 ReallyDo Varchar2(200);
Begin

 if ( NVL(para_BusinessId,' ') = ' ' ) then
 raise_application_error(-20001, '输入的单位或者商品不能为空');
 rollback;

 elsif ( NVL(para_BusinessId,' ') <> ' ' ) then
 select case when exists ( SELECT 1 FROM contactdoc aa
 JOIN GOODSATTR cc on cc.goodsid = para_GoodsId
 WHERE INSTR(',' || LTRIM(RTRIM(aa.conscope)) || ',', ',' || LTRIM(RTRIM(cc.formula)) || ',') = 0
 AND aa.contype = 0 and aa.isyw='Y' and aa.businessid = para_BusinessId ) then '当前单位的委托书类型为：剂型 ，但是当前选中的商品的剂型不在此范围类请注意'
 when exists ( SELECT 1 FROM contactdoc aa
 JOIN GOODSATTR cc on cc.goodsid = para_GoodsId
 WHERE INSTR(',' || LTRIM(RTRIM(aa.conscope)) || ',', ',' || LTRIM(RTRIM(cc.gcategory)) || ',') = 0
 AND aa.contype = 2 and aa.isyw='Y' and aa.businessid = para_BusinessId) then '当前单位的委托书类型为：商品类别 ，但是当前选中的商品的类别不在此范围类请注意'
 when exists ( SELECT 1 FROM contactdoc aa
 WHERE INSTR(',' || LTRIM(RTRIM(aa.conscope)) || ',', ',' || LTRIM(RTRIM(para_GoodsId)) || ',') = 0
 AND aa.contype = 1 and aa.isyw='Y' and aa.businessid = para_BusinessId ) then '当前单位的委托书类型为：商品 ，但是当前选中的商品不在此范围类请注意'
 ELSE '正常' END
 Into ReallyDo
 from DUAL;
 end if ;
 Return(ReallyDo);

End GetWtsStatus;
