## 问题原因

## 解决方案

零售业务开票的拦截函数uf_saveCheck中增加代码：
 // 拦截重复录入追溯码
 //第一次明细加载完后
 var items ={};
 for(var i = 0 ; i < ds_imieHis.recordCount; i++) {//循环明细数据集
 if (ds_imieHis.getValueAt(i, "imeino") in items) {//如果集合里面存在该编号的商品则在该商品的total属性上进行累加
 items[ds_imieHis.getValueAt(i, "imeino")].num = 1;
 } else {//如果不存在则新建这个对象
 items[ds_imieHis.getValueAt(i, "imeino")] = {num : 0};
 }
 }
 // 遍历对象集合 
 for (var obj in items) {
 if (items[obj].num > 0 ) {
 var row = ds_imieHis.find("imeino = '"+obj+"'");
 if(row != -1){
 await form.alert("当前销售存在重复录入追溯码，请检查！");
 return false;
 }
 }
 }
