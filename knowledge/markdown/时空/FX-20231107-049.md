## 问题现象

客户需要同时打印随货同行联、质检报告单、电子面单，常规方案是使用一键打印程序完成，但如果客户要求根据不同业务类型，打印不同样式的随货同行联，则一键打印程序处理不了

## 问题原因

一键打印不能设置多个清单打印方案，并根据不同条件自动选择打印哪一个方案

## 解决方案

随货同行联打印采用智友自带打印，其他打印通过智友调用一键打印程序完成。
使用以下函数调用一键打印程序
function uf_print_get(billcode){

 var url = "http://127.0.0.1:18090?fhdjbh="+billcode+"&copys=2";
 var time = 1000;
 var timeout = false;
 var request = new XMLHttpRequest();
 
 var timer = setTimeout(function () {
 timeout = true;
 request.abort();
 }, time);
 
 request.open("GET", url);
 
 request.onreadystatechange = function () {
 if (request.readyState !== 4) {
 return;
 }
 if (timeout) {
 //alert("timeout");
 return;
 }
 clearTimeout(timer);
 if (request.status === 200) {
 //callback(request.responseText);
 //alert(request.responseText);
 }
 };
 
 request.send(null);
 
}
