## 解决方案

function uf\_ip()\{
 var wshNetwork = new ActiveXObject("WScript.Network");
 var locator = new ActiveXObject("WbemScripting.SWbemLocator");
 var service = locator.ConnectServer(".");
 var properties = service.ExecQuery("SELECT \* FROM Win32\_NetworkAdapterConfiguration Where IPEnabled=TRUE");
 var e = new Enumerator(properties);
 for (; !e.atEnd(); e.moveNext()) \{
 var p = e.item();
 if (p.IPAddress == null) \{
 continue;
 \}
 var ipinfo = p.IPAddress(0);
 \}
 var ComputerName = wshNetwork.ComputerName;
 var info = ipinfo \+ "\|\|" \+ ComputerName;
// alert(info);
 ipinfo = ipinfo.replace('.','*');
 ipinfo = ipinfo.replace('.','*');
 ipinfo = ipinfo.replace('.','\_');
 return ipinfo;
\}
