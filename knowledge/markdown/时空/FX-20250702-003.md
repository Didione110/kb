## 解决方案

function uf_scancode() {
 textbox1.setValue("");
 var fileInput = document.createElement('input');
 fileInput.type = 'file';
 fileInput.style.display = 'none';
 fileInput.accept = '.xml';
 document.body.appendChild(fileInput);
 fileInput.addEventListener('change', function() {
 if (this.files && this.files[0]) {
 var file = this.files[0];
 var isXml = file.name.toLowerCase().endsWith('.xml');
 if (isXml) {
 var reader = new FileReader();
 reader.onload = async function(e) {
 var fileContent = e.target.result; 
 var jsonObj = xmlToJson(fileContent);
 var jsonString = JSON.stringify(jsonObj, null, 2);
 await uf_load_xmlmx(jsonString)
// console.log('JSON结果:', jsonString);
 };
 reader.readAsText(file); // 读取为文本，也可改用 readAsBinaryString 等
 textbox1.setValue(file.name); // 只显示文件名，更友好
 } else {
 form.alert("请选择XML格式的文件！");
 textbox1.setValue("");
 }
 }
 document.body.removeChild(fileInput);
 });
 
 fileInput.click();
}
