## 问题原因

退补价单按金额进行退补，需要人为计算分摊到原单明细中，除不尽的情况下导致客户不知道该如何填写

## 解决方案

直接填写折扣金额，系统根据金额比例自动进行分摊，js代码如下：
function uf_dedamt(){
 var zkamt = dshz.field("k_dedamt").value;
 var taxamount = dshz.field("taxamount").value;
 var ft_TaxAmount = 0;
 for(var k = 0; k < dsmx.recordCount;k++){
 //分摊逻辑：分摊比例=明细金额/总金额
 dsmx.setValueAt(k,"TaxAmount",Utility.formatFloat(zkamt * Utility.formatFloat(dsmx.getValueAt(k,"OldTaxAmount") / taxamount,2),2));
 dsmx.setValueAt(k,"TaxPrice",Utility.formatFloat(dsmx.getValueAt(k,"TaxAmount") / dsmx.getValueAt(k,"Num"),3));
 dsmx.setValueAt(k,"Price",Utility.formatFloat(dsmx.getValueAt(k,"TaxAmount") / dsmx.getValueAt(k,"Num"),3) / (1 + dsmx.getValueAt(k,"Rate") / 100));
 dsmx.setValueAt(k,"BaseTaxPrice",Utility.formatFloat(dsmx.getValueAt(k,"TaxAmount") / dsmx.getValueAt(k,"Num"),3));
 dsmx.setValueAt(k,"BasePrice",Utility.formatFloat(dsmx.getValueAt(k,"TaxAmount") / dsmx.getValueAt(k,"Num"),3) / (1 + dsmx.getValueAt(k,"Rate") / 100));
 dsmx.setValueAt(k,"Amount",Utility.formatFloat(dsmx.getValueAt(k,"Price") * dsmx.getValueAt(k,"Num")))
 dsmx.setValueAt(k,"NewTaxPrice",Utility.formatFloat(dsmx.getValueAt(k,"OldTaxPrice") + dsmx.getValueAt(k,"TaxPrice")));
 dsmx.setValueAt(k,"NewPrice",Utility.formatFloat(dsmx.getValueAt(k,"OldPrice") + dsmx.getValueAt(k,"Price")));
 dsmx.setValueAt(k,"NewBaseTaxPrice",Utility.formatFloat(dsmx.getValueAt(k,"OldTaxPrice") + dsmx.getValueAt(k,"TaxPrice")));
 dsmx.setValueAt(k,"NewBasePrice",Utility.formatFloat(dsmx.getValueAt(k,"OldPrice") + dsmx.getValueAt(k,"Price")));
 dsmx.setValueAt(k,"Tax",dsmx.getValueAt(k,"TaxAmount") - dsmx.getValueAt(k,"Amount"));
 }

 //为避免分摊后的金额与输入金额有差异，差异强制计算到第一条
 for(var j = 0; j < dsmx.recordCount;j++){
 ft_TaxAmount += dsmx.getValueAt(j,"TaxAmount");
 }
 if(ft_TaxAmount != zkamt){
 var diff = zkamt - ft_TaxAmount;
 dsmx.setValueAt(0,"TaxAmount",dsmx.getValueAt(0,"TaxAmount") + diff);
 }
}
