ALTER PROCEDURE [dbo].[SKZY_ZLDR]
AS --资料导入
BEGIN
 DECLARE @ENTID VARCHAR(20)
 SET @ENTID = ( SELECT TOP 1
 entid
 FROM OPENQUERY(SKZY , 'SELECT a.entid FROM ENTDOC a where a.beactive=''Y'' ') )
 
 
 DECLARE @ORGID VARCHAR(20)
 SET @ORGID = ( SELECT TOP 1
 orgid
 FROM OPENQUERY(SKZY , 'SELECT a.orgid FROM ORGDOC a where a.beactive=''Y'' ') )
 
 
 -- 商品资料导入
 INSERT OPENQUERY(SKZY ,
 'SELECT ENTID , GOODSID , ISNEW , ISFREEZE , ISABANDON , BEACTIVE , GOODSCODE , OLDCODE , GOODSNAME , LOGOGRAM , GOODSSPEC , PLACE , MANUFACTURER , 
 CREATETIME , CREATERID , CREATERINFO , LASTMODIFYTIME , ISPASS , ISCOMMPASS , ISCOMBIN , ISIMPORTED , SaleManId ,BarCode from GoodsDoc')
 SELECT @ENTID AS ENTID , RTRIM(a.spid) AS GOODSID , 'N' AS ISNEW , 'N' AS ISFREEZE , 'N' AS ISABANDON , 'Y' AS BEACTIVE ,
 RTRIM(a.spbh) AS GOODSCODE , RTRIM(a.spbh) AS OLDCODE , a.spmch AS GOODSNAME , a.zjm AS LOGOGRAM , a.shpgg AS GOODSSPEC ,
 a.shengccj AS PLACE , a.shpchd AS MANUFACTURER , CONVERT(CHAR(19) , GETDATE() , 121) AS CREATETIME , 'K3UL56BUG91' /*管理员*/ AS CREATERID ,
 'o=O3UL56BUG91,d=K3YUFN7HGLU,u=K3UL56BUG91,' AS CREATERINFO , CONVERT(CHAR(19) , GETDATE() , 121) AS LASTMODIFYTIME , 'N' AS ISPASS ,
 'N' AS ISCOMMPASS , 'N' AS ISCOMBIN , 'Y' AS ISIMPORTED ,
 ( SELECT TOP 1
 staffid
 FROM OPENQUERY(SKZY , 'SELECT a.staffid FROM staffdoc a where a.beactive=''Y'' and a.staffname = ''+a.gsp_cgzy+'' ') ) AS SaleManId,a.sptm
 FROM ksoa..spkfk a
 WHERE a.zhengzzt = 1
 AND a.spbh NOT IN ( SELECT goodscode
 FROM OPENQUERY(SKZY , 'SELECT a.goodscode FROM goodsdoc a ') )
