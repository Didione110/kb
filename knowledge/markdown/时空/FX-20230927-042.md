## 问题现象

智友系列陈年老BUG问题
在修改客户卡片信息时，有多个数据集合，如证照。委托书。地址，等
标准操作方法：修改完客户信息保存后，再去委托书修改发起，再去证照修改发起，再去地址修改发起
问题：操作异常麻烦，几乎所有的客户都抱怨过

## 问题原因

标准方案无对应解决方案

## 解决方案

魔改代码
在表单save的时候，同时发起N个工作流，记录修改审批的内容
然后审批后生效，案例代码如下
//判断证照列表是否做过数据修改,如果修改,则手动发起修改工作流
 if (form.parameters.action == "modify" && typeof form.args.isSetPass == "undefined") {
 var zzmxxgzt = "N";//初始化默认值给N
 for (var i = 0; i < ds_djmx.recordCount; i++) { //无需翻译经营范围，取消
 if(ds_djmx.getRowState(i) != 0){
 zzmxxgzt = "Y";
 }
 }
 //记录存在修改的时候 发起工作流 记录证照修改历史 
 if(zzmxxgzt == "Y"){
 var zzmxxgbillno = FormUtil.getIdentity("billno", "", "") ;
 var zzmxxgbillcode = FormUtil.getIdentity("billcode", "61ntc7e0a0kgsljr", "") ;
 for (var i = 0; i < ds_djmx.recordCount; i++) { 
 ds_djmx.setValueAt(i,"billno",zzmxxgbillno);
 ds_djmx.setValueAt(i,"billsn",i+1);
 }
 ds_djhz.openEmpty();
 ds_djhz.append();
 ds_djhz.setValue("EntId",form.userinfo.entid);
 ds_djhz.setValue("billno",zzmxxgbillno);
 ds_djhz.setValue("billcode",zzmxxgbillcode);
 ds_djhz.setValue("RuleId","61ntc7e0a0kgsljr");
 ds_djhz.setValue("Dates",DateUtil.curDate());
 ds_djhz.setValue("SysDates",DateUtil.getServerDate());
 ds_djhz.setValue("OrgId",form.userinfo.orgid);
 ds_djhz.setValue("DeptId",form.userinfo.deptid);
 ds_djhz.setValue("BusinessId",DsMain.field("BusinessId").value);
 ds_djhz.setValue("OnTime",DateUtil.curTime());
 ds_djhz.setValue("caozy",form.userinfo.staffid);
 ds_djhz.setValue("Summaries","客户证照资质变更");
 ds_djhz.setValue("Remark","证照更新"); 
 ds_djhz.setValue("ChangeType","khzz"); 
 form.saveDatasets(ds_djhz,ds_djmx);//数据存入变更历史 
 //存盘后发起工作流审批
 var par = {};
 par.v_billno = zzmxxgbillno;
 par.v_billcode = zzmxxgbillcode;
 par.summaries = form.userinfo.staffname+","+DsMain.field("BusinessName").value ;
 par.remark = "证照更新"
 par.wfid = "45jj4cw40mweepmb";
 par.ruleid = "61ntc7e0a0kgsljr";
 par.formid = "4sdxxamp58kczekn";
 FormUtil.executeServerScript("srv_createWorkFlow",par,ds_djhz);
 }
 }
function srv_createWorkFlow(par){
 var v_billno = par.v_billno ; 
 var v_billcode = par.v_billcode;
 var summaries = par.summaries
 var remark = par.remark ;
 var wfid = par.wfid ;
 var formid = par.formid ;
 var ruleid = par.ruleid ;
 FormUtil.addBillReviewItem(formid, v_billno); 
 FormUtil.startWorkflow(v_billno, v_billcode, wfid, 'save', ruleid, summaries ,remark )
 
}
