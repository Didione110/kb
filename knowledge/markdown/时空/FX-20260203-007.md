## 问题现象

部分进口药品供应商名称存在英文 ' 号如Dr.Reddy's Laboratories Limited
在将其作为查询条件时会出现SQL command not properly ended提示无法查询

## 问题原因

'号截断了语句导致查询出错

## 解决方案

js中处理'为''在数据库中是等价的再进行查询可以正常筛选
function escapeSingleQuoteForOracle(str) {
 // 1. 处理空值/非字符串情况，避免报错
 if (str === null || str === undefined || typeof str !== 'string') {
 return '';
 }

 // 2. 初始化结果字符串
 var result = '';
 
 // 3. 遍历每个字符，逐个判断拼接
 for (var i = 0; i < str.length; i++) {
 //const char = str[i]; // 用charAt兼容老旧环境（替代str[i]）
 var char = str.charAt(i);
 // 如果是单引号，拼接两个；否则拼接原字符
 if (char == "'") {
 result += "''";
 } else {
 result += char;
 }
 }

 return result;
}

var Manufacturer = escapeSingleQuoteForOracle(ds_djhz.field("Manufacturer").value)
params.Manufacturer = "%" + Manufacturer + "%";
