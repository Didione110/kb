## 问题现象

目前的智友V10 V11 V12标准方案均无法拦截必备证照

## 问题原因

厂家就没考虑到

## 解决方案

1：维护完不同客户类型或不同供应商的类型的必备证照 
2：在首营录入资料页面，获取列表
3：存盘save\_check进行拦截 ，代码案例参照如下

function uf\_arraysContain(arr1, arr2) \{
 return arr2.every(function(item) \{
 return arr1.includes(item);
 \});
\}
```
var needcert_list ="";//获取系统维护的必备证照列表
var par = {};
par.rfcate = DsMain.field("ClientType").value;
needcert_list = DBUtil.uniqueValue("uf_sql_getneedcert_list",par)

var now_needcert_list ="";//校验当前录入的证照信息有没有瞎搞,该填的必须要填
for (var i = 0; i < ds_djmx.recordCount; i++) {
 if(!Utility.isSpace(ds_djmx.getValueAt(i, "CertName"))){
 now_needcert_list = now_needcert_list+ ds_djmx.getValueAt(i, "CertName") + ",";
 }
}
now_needcert_list = (now_needcert_list.substr(0, now_needcert_list.length - 1));//证照列表
var arr1_text = ","+now_needcert_list+",";
var arr2_text = ","+needcert_list+"," ;
var arr1 = arr1_text.split(",");
var arr2 = arr2_text.split(",");
var arr_diff = arr2.filter(item => !arr1.includes(item));
if (!uf_arraysContain(arr1, arr2)){
 form.alert("当前客户需要录入的证照有:【"+needcert_list+" 】 当前证照信息未达到质管要求,还缺少：【"+arr_diff+"】 请注意");
 return false;
}

```
