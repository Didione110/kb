## 问题现象

## 问题原因

此处类似抄送，不需要真实审批

## 解决方案

不动菜单的情况下新增工作流，复制出流程id以及单据的ruleid,替换下语句，此处使用采购入库单

function uf_close() {
 try {
 var wf;
 if (WorkFlow.formMenuId.indexOf(".") > -1) {
 var ids = WorkFlow.formMenuId.split(".");
 wf = new WorkFlow(ids[0], ids[1]);
 } else {
 wf = new WorkFlow(WorkFlow.formMenuId);
 }
 
 var wfInfo = wf.buildWfInfo();
 if (wfInfo && typeof wfInfo === "object" && wfInfo.then) {
 form.alert("系统不支持异步工作流，请联系管理员");
 return;
 }
 
 var optionXml = "<option"
 + " moduleid=\"" + form.parameters.moduleid + "\""
 + " orgno=\"" + form.userinfo.orgno + "\""
 + " formid=\"" + form.formid + "\""
 + " keyfield=\"" + form.parameters.keyfield + "\""
 + " idprefix=\"" + form.parameters.idprefix + "\""
 + " billno=\"" + form.billno + "\""
 + " isworkflow=\"true\""
 + " wfinfo=\"" + Utility.xmlEncode(Utility.toJSONString(wfInfo)) + "\""
 + "/>";
 
 var service = "workflow.forward";
 var url = form.parameters.contextpath + "/formservice?service=" + service + form.serviceprefix;
 
 var xhr;
 if (typeof XMLHttpRequest !== "undefined") {
 xhr = new XMLHttpRequest();
 } else if (typeof ActiveXObject !== "undefined") {
 xhr = new ActiveXObject("Microsoft.XMLHTTP");
 } else {
 form.alert("浏览器不支持XMLHttpRequest");
 return;
 }
 
 xhr.open("POST", url, false);
 xhr.send(optionXml);
 
 // 延时 500ms 关闭表单，让服务端处理完
 setTimeout(function() {
 form.close();
 }, 500);
 
 } catch (e) {
 form.alert("待办关闭失败：" + e.message);
 }
}

async function uf_close() {
 console.log('开始执行待办关闭...');
 
 try {
 var wfInfo = WorkFlow.wfInfo;
 if(!wfInfo) {
 // 如果 wfInfo 不存在，需要先构建
 var ids, wf;
 if(WorkFlow.formMenuId.indexOf(".") > -1) {
 ids = WorkFlow.formMenuId.split(".");
 wf = new WorkFlow(ids[0], ids[1]);
 } else {
 wf = new WorkFlow(WorkFlow.formMenuId);
 }
 wfInfo = await wf.buildWfInfo();
 WorkFlow.wfInfo = wfInfo;
 }
 
 // 构建 option XML
 var buf = [];
 buf.push('<option');
 buf.push(' moduleid="' + form.parameters.moduleid + '"');
 buf.push(' orgno="' + form.userinfo.orgno + '"');
 buf.push(' formid="' + form.formid + '"');
 buf.push(' keyfield="' + form.parameters.keyfield + '"');
 buf.push(' idprefix="' + form.parameters.idprefix + '"');
 buf.push(' billno="' + form.billno + '"');
 buf.push(' isworkflow="true"');
 buf.push(' wfinfo="' + Utility.xmlEncode(Utility.toJSONString(wfInfo)) + '"');
 buf.push('/>');
 var optionXml = buf.join('');
 
 // 直接发请求
 var service = "workflow.forward";
 var url = form.parameters.contextpath + "/formservice?service=" + service + form.serviceprefix;
 var result = WebService.directCall(url, optionXml, "POST", null);
 
 console.log('✅ 待办关闭成功！');
 await form.close();
 
 } catch (e) {
 console.error('❌ 失败：', e.message);
 }
}
