## 问题现象

智友全系类产品，表单的复制权限默认是没开启的，而且只能在设计器里面开启
如果客户需要对不通的操作员进行设置，是不可以的

## 问题原因

北京智友没考虑到这一块

## 解决方案

在所有的表单的onload里引用加载获取表单复制权限
1：制作人员表单权限复制表
create table copyAuth(entid ,userid,formid)
2：制作公共函数判断人员是否有表单复制的权限
function srv\_getCopyAuth() \{
 //userid:操作员ID
 //formid:传入的表单ID
 var sql = "select formid from CopyAuth "
 \+ " where formid='" \+ param1 \+ "' and userid = '" \+ param2 \+ "'";
 var ret = DBUtil.select(sql);
 if (ret.values.length \> 0)\{
 return 1;
 \}
 else \{
 return 0;
 \}
\}

3：在所有表单上引用此方法获取对应属性的设置
```
//获取当前操作员是否有表单复制的权限
var param = {};
param.moduleid="2k7rp7e252ww5bkn";
param.funcName="srv_getCopyAuth";
var subparam={};
subparam.param1 = form.id;
subparam.param2 = form.userinfo.userid;
param.param = subparam;
var result = WebService.call("exceFuction", param, true); 
if( result == 1 ){ // 判断当前操作员的表单复制权限
 for (var i = 0 ; i < form.auxobjs.datagrids.length ; i++) { 
 eval(form.auxobjs.datagrids[i]).enablecopy = "true"; 
 }
} else { //如果没有则返回false不让复制
 for (var i = 0 ; i < form.auxobjs.datagrids.length ; i++) { 
 eval(form.auxobjs.datagrids[i]).enablecopy = "false"; 
 } 
}
//更新页面数据集合
var length = form.auxobjs.datasetobjects.length > form.auxobjs.datasets.length ? form.auxobjs.datasetobjects.length : form.auxobjs.datasets.length;
for (var i = 0 ; i < length ; i++) { 
 if (form.auxobjs.datasetobjects.length > i){
 eval(form.auxobjs.datasetobjects[i]).updateGrid();
 }
 if (form.auxobjs.datasets.length > i){
 eval(form.auxobjs.datasets[i]).updateGrid();
 }
} 

```
