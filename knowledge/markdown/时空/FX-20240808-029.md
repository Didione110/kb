## 问题现象

早期的时空智友V10 V11 V12 针对基础资料修改，是通过pf_dataModRec函数进行对比更新生成历史记录，但是存在的问题是，只能通过单数据集单行的数据进行对比，涉及到grid多行数据时即无法比较

## 问题原因

早期智友无此问题的相关经验，在最近的dherp的root里已经解决此问题，新增了pf_dataModRec_all函数

## 解决方案

基础资料修改以及单据修改的表单中，遵循以下代码原则
首先在表单加载前，定义dataset的模型
//keyFlds 是主键，preFlds 是监测修改内容的字段名称 ，ignoreFlds是忽略字段
var datasets = [
 {dataset: DsMain, keyFlds : "BusinessId,EntId", preFlds : "",ignoreFlds:"BusinessCont,LastStfId"},
 {dataset: ds_djmx, keyFlds : "CertName", preFlds : "CertName,CertCode,AuthDate,ValDate"},
 {dataset: ds_wtsmx, keyFlds : "ContactId", preFlds : "ConsCert,Contact,Telephone,IdCard,IdCardDate,ConsCertDate,ConType,ConScope", ignoreFlds:""},
 {dataset: ds_address, keyFlds : "AddressId", preFlds : "Address,k_province,k_city,k_area", ignoreFlds:""}
 ];

onload加载时
 if (form.parameters.action == "modify" && regLog == "Y") {
 if (typeof form.args.isSetPass != "undefined") { //工作流审批修改的时候
 DsMain.openEmpty();
 ds_djmx.openEmpty();
 ds_wtsmx.openEmpty();
 ds_address.openEmpty();
 var jsonArr = await form.args.jsonArr; 
 await pf_dataModRec_all(2, datasets, jsonArr, {}, "", 0);
 } else {
 jsonArrOld = await pf_dataModRec_all(1, datasets, {}, {}); //正常情况下首次打开的时候是这样的
 }
 }
save存盘时
if (form.parameters.action == "modify" && regLog == "Y" && typeof form.args.isSetPass == "undefined") {
jsonArrNew = await pf_dataModRec_all(1, datasets, {}, {});

 await pf_dataModRec_all(4, datasets, jsonArrOld, jsonArrNew, businessid, 1, parMod);

}
