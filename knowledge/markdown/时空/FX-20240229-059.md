## 问题现象

在SQLSERVER数据库的时空体系产品中，客户在上基础资料的时候，可能会存在各种不确定因素，会把换行符录入至系统中，造成在数据库查询的时候，字段值的尾巴存在空格，在导出excel数据的时候，造成字段和数据错位，而且在sql语句中用rtrim和ltrim是无法去掉对应的空格

## 问题原因

原因:中间存在回车符或者换行符，所以要先将此符号替换掉；
https://www.cnblogs.com/cuozhixu/p/7285652.html
单独处理字段数据的办法在此链接内，然后发散思维，目前这种问题只有在遇到的时候才知道，如何在事前就避免呢？ 为此我写了一份T-sql，遍历数据库进行查找，然后再来批量处理，代码如下

## 解决方案

IF OBJECT_ID('tempdb..#tmp_table') IS NOT NULL
 BEGIN
 DROP TABLE #tmp_table;
 END;

 SELECT ROW_NUMBER() OVER ( ORDER BY a.TBNAME ) AS tms , tbname , tbdesc --, b.RowCounts
 INTO #tmp_table
 FROM tblist a
 WHERE ( a.moduleid <> '007nfmq38ycx30dv' )
 AND a.attr = 'Y'
 AND objecttype = 'T'
 AND a.tbname IN ( SELECT tbname
 FROM tbstru
 WHERE fdname = 'EntId' )
 AND a.TBNAME IN ( 'Goodsdoc' , 'businessdoc' )

 DECLARE @count INT;
 SET @count = 1;

 DECLARE @tmscount INT
 SET @tmscount = ( ( SELECT COUNT(1)
 FROM tblist a
 WHERE ( a.moduleid <> '007nfmq38ycx30dv'
 AND a.moduleid IN ( SELECT moduleid
 FROM moduledoc
 WHERE developerid = 'E061W25SK3M' )
 )
 AND a.attr = 'Y'
 AND objecttype = 'T'
 AND a.tbname IN ( SELECT tbname
 FROM tbstru
 WHERE fdname = 'EntId' ) 
 ) )
 DECLARE @sql_text NVARCHAR(MAX);
-遍历流水表
 WHILE @count <= @tmscount
 BEGIN
 DECLARE @mxcount INT;
 SET @mxcount = 1;
 --取出当前循环的表里字段的数量
 DECLARE @mxtmscount INT
 SET @mxtmscount = ( SELECT COUNT(1)
 FROM INFORMATION_SCHEMA.COLUMNS
 WHERE TABLE_NAME = ( SELECT TOP 1 aa.tbname FROM #tmp_table aa WHERE aa.tms = @count ) )
 --遍历表内字段中存在问题的字段
 WHILE @mxcount <= @mxtmscount
 BEGIN
 DECLARE @output_tms INT
 
 SET @sql_text = N'SELECT @output_tms = count(1) from ' + ( SELECT TOP 1 aa.tbname FROM #tmp_table aa WHERE aa.tms = @count )
 + ' where ' + ( SELECT COLUMN_NAME
 FROM INFORMATION_SCHEMA.COLUMNS
 WHERE TABLE_NAME = ( SELECT TOP 1 aa.tbname FROM #tmp_table aa WHERE aa.tms = @count )
 AND ORDINAL_POSITION = @mxcount ) + ' <> LTRIM(RTRIM(REPLACE(REPLACE(REPLACE('
 + ( SELECT COLUMN_NAME
 FROM INFORMATION_SCHEMA.COLUMNS
 WHERE TABLE_NAME = ( SELECT TOP 1 aa.tbname FROM #tmp_table aa WHERE aa.tms = @count )
 AND ORDINAL_POSITION = @mxcount ) + ',char(13),''''),char(10),''''),char(09),''''))) '
 
 EXEC sp_executesql @sql_text , N'@output_tms NVARCHAR(1000) OUTPUT' , @output_tms OUTPUT
 
 IF @output_tms > 0
 BEGIN
 
 PRINT @output_tms
 
 PRINT @sql_text
 END 
 
 SET @mxcount = @mxcount + 1;
 END 
 
 SET @count = @count + 1;
 END;
