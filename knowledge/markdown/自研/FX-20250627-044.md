## 解决方案

private static PdfPTable createMainTable2(List<List<String>> dataDetail) throws IOException, DocumentException, com.itextpdf.text.DocumentException {
 // 主表格（列数根据表头定义）
 PdfPTable table = new PdfPTable(dataDetail.get(0).size());
 table.setWidthPercentage(100); // 宽度占满列宽
 table.setHorizontalAlignment(Element.ALIGN_LEFT);

 // 添加表头和数据
 for (List<String> row : dataDetail) {
 for (String cellValue : row) {
 PdfPCell cell = new PdfPCell(new Phrase(cellValue, getChineseFont()));
 cell.setMinimumHeight(30); // 最小行高
 cell.setPaddingLeft(15); // 左侧内边距15像素
 cell.setBorder(Rectangle.NO_BORDER); // 隐藏该单元格所有边框
 cell.setPadding(5);
 table.addCell(cell);
 }
 }
 return table;
 }
