## 问题现象

明细表通过公式定义器不生效

## 问题原因

低代码语法不一样，明细表计算是list算法，产品未考虑，需要单独增加JS

## 解决方案

增加JS片段，并对计算控件进行动态监控，然后循环计算后进行赋值
<xform:editShow>
<script>
// 质量
AttachXFormValueChangeEventById("fd_3c8b48b6cf7e1c",changeGrade);
// 价格
AttachXFormValueChangeEventById("fd_3c8b48b846da94",changeGrade);
// 安全
AttachXFormValueChangeEventById("fd_3c8b48b9bde692",changeGrade);
// 履约
AttachXFormValueChangeEventById("fd_3c8b48bae514fc",changeGrade);
// 售后
AttachXFormValueChangeEventById("fd_3c8b48c266d02e",changeGrade);

function changeGrade(){
 // 获取明细表行数
 const length = $("#TABLE_DL_fd_3c8b44b27d1194").children().first().children().length-3;
 // 遍历明细表行数
 for(var i = 0;i < length; i++) {
 const scoreStr = GetXFormFieldValueById("fd_3c8b44b27d1194." + i + ".fd_3c8b48c69517d2")[0];
 const score = parseInt(scoreStr);
 if(score >= 85){
 SetXFormFieldValueById("fd_3c8b44b27d1194." + i + ".fd_3c8b49e7321074", "合格");
 } else {
 SetXFormFieldValueById("fd_3c8b44b27d1194." + i + ".fd_3c8b49e7321074", "不合格");
 }
 }
}

</script>
</xform:editShow>
