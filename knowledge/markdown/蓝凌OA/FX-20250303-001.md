## 问题现象

需要判断文本框只能输入一个名字

## 问题原因

不能同时代发起多人，负责不方便核算

## 解决方案

项目可以参照如下代码进行修改，但是同时也有边界如输入"张三李四"是有效的，如输入"张三、李四"则是无效的
<script>
//监听文本框变化
AttachXFormValueChangeEventById("fd_3d8203385c1b7a",changeGrade);

function isValidName(name) {
 // 如果输入为空，直接返回false
 if (!name || name.trim() === "") return false;
 const nameRegex =/^[\u4e00-\u9fa5A-Za-z'-]+$/;

 return nameRegex.test(name.trim());
}

// 事件处理函数
function changeGrade() {
 try {
 // 获取指定ID的文本框值
 const nameInput = GetXFormFieldValueById("fd_3d8203385c1b7a")[0];

 // 检查是否为有效名字
 if (isValidName(nameInput)) {
 SetXFormFieldValueById("fd_3dd9fefb2c5d80", "有效");
 } else {
 SetXFormFieldValueById("fd_3dd9fefb2c5d80", "无效");
 }
 } 
}

</script>
