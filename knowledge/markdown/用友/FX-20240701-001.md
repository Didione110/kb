## 问题现象

## 解决方案

SELECT
 bu1.code as cjr,
 apb.id AS djiderp,-- 关联子表
 apb.DocNo AS dherp,-- 单号
 apb.PayRFDate AS sqrqerp,-- 申请日期
 apb.ExpectPayDate AS yjfkrerp,-- 预计付款日
 apb.RequestObjType AS qkdxerp,-- 请款对象 客户 0,部门 2，员工 3， 其它 4，供应商 1
-- apb.ReqFundUse as ReqFundUse,-- 明细中获取 请款用途 保证金 0, 杂项 4, 预付款 1, 质保金 3, 标准 2
 bct.Name AS bzerp,-- 币种
 bot1.name AS lyzzerp,-- 来源组织
 bot2.name AS jszzerp,-- 结算组织
 cot.Code AS ywyerp,-- 业务员
 apb.PayMode AS fkfserp,-- 付款方式
 apb.IsCashierConfirmed AS cnqrerp,-- 出纳确认
 apbt.Memo AS beizerp,-- 备注
 bu.code AS fqrerp,-- 发起人编码
-- 明细

 
 apru.LineNum AS xherp,-- 行号
 apru.Supp_Code AS gysbmerp,-- 供应商
 cst.name AS gysmcerp,-- 供应商名称
 csst.Name AS gyswzerp,-- 供应商位置
 apru.Cust_code AS khccerp,-- 客户产出
 apru.CustSite_Code AS khwzerp,-- 客户位置位置
 apru.RFTotalMoney AS zkqqkjeerp,-- 折扣前请款金额
 CAST ( apru.RFDisCount AS DECIMAL ( 10, 2 ) ) AS qkdyzkerp,-- 请款对应折扣
 apru.RFPayMoney AS zkhqkjeerp,-- 折扣后请款金额
 apru.AcmPayFCMoney AS fkbbjeerp,-- 付款本币金额
 bot4.name AS ywzzerp,-- 业务组织
 bot5.name AS lyzzerp,-- 来源组织
 csmt.name AS jsfserp,-- 结算方式
 apru.OppAccBkAccount AS skyxzherp,-- 收款银行账号
 apru.RecBkAccName AS skzhmcerp,-- 收款账户名称
 apru.OppAccBk AS khyxerp,-- 开户银行
 cot2.code AS ywyerp,-- 业务员
 apru.Item_ItemCode AS lpbmerp,-- 料品编码
 cp.code AS xmbmerp,-- 项目编码
 cp2.name AS xmmcerp, -- 项目名称
 apru.ReqFundUse as qkyterp, -- 请款用途
 
 case 
when aprfd.SrcBillType = 1 then (select STRING_AGG(DocNo, ',') from PM_PurchaseOrder ppo where ppo.id = aprfd.SrcBillID)
when aprfd.SrcBillType = 0 then ( SELECT STRING_AGG(aaa.DocNo, ',') FROM AP_APBillHead aaa where aaa.id = aprfd.SrcBillID)
when aprfd.SrcBillType = 3 then (select STRING_AGG(DocNo, ',') from PM_Receivement ppo where ppo.id = aprfd.SrcBillID)
end as yfddherp,-- 应付单单号
case 
when aprfd.SrcBillType = 1 then (select STRING_AGG(ppl.ItemInfo_ItemCode, ',') from PM_PurchaseOrder ppo left join PM_POLine ppl on ppo.id = ppl.PurchaseOrder where ppo.id = aprfd.SrcBillID)
when aprfd.SrcBillType = 0 then (select STRING_AGG(aaaa.Item_ItemCode, ',') from AP_PayReqFundDetail aaaa where apru.id = aaaa.PayReqFundUse)
when aprfd.SrcBillType = 3 then (select STRING_AGG(ppr.ItemInfo_ItemCode, ',') from PM_Receivement ppo left join PM_RcvLine ppr on ppo.id = ppr.Receivement where ppo.id = aprfd.SrcBillID)
end as yfdlpbmerp-- 应付单料品编码

 FROM
 AP_PayReqBillHead apb
 left join base_user bu1 on bu1.name = apb.createdby
 LEFT JOIN Base_Currency_trl bct ON apb.ReqFundPC = bct.id
 LEFT JOIN Base_User bu ON apb.createdby = bu.name
 LEFT JOIN Base_Organization_trl bot1 ON apb.SrcBizOwnerOrg = bot1.id
 LEFT JOIN Base_Organization_trl bot2 ON apb.SettleOrg = bot2.id
 LEFT JOIN CBO_Operators cot ON cot.id = apb.Transactor
 LEFT JOIN AP_PayReqBillHead_trl apbt ON apb.id = apbt.id
 -- 明细
 LEFT JOIN AP_PayReqFundUse apru ON apb.id = apru.PayReqFundHead
 LEFT JOIN CBO_SupplierSite_trl csst on csst.id = apru.SuppSite_SupplierSite
 LEFT JOIN AP_PayReqFundDetail aprfd ON apru.id = aprfd.PayReqFundUse
 LEFT JOIN AP_APBillHead aah ON aah.id = aprfd.SrcBillID
 LEFT JOIN Base_Organization_Trl bot4 ON apru.BizOrg = bot4.id
 LEFT JOIN Base_Organization_Trl bot5 ON apru.SrcOrg = bot5.id
 LEFT JOIN CBO_SettlementMethod_trl csmt ON apru.SttlMethod = csmt.id
 LEFT JOIN CBO_Operators cot2 ON cot2.id = apru.Transactor
 LEFT JOIN CBO_Department_trl cdt2 ON apru.Dept = cdt2.id
 LEFT JOIN CBO_Project cp ON apru.project = cp.id
 LEFT JOIN CBO_Project_trl cp2 ON apru.project = cp2.id
 LEFT JOIN CBO_SUPPLIER_trl cst ON cst.id = apru.Supp_Supplier
