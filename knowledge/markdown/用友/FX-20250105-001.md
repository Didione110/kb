## 解决方案

需要执行以下语句才可以还原开立状态的票据
--删除单据TOGL表
delete from NM_NMToGL_Trl where ID=(select ID from NM_NMToGL where DocID=(select ID from NM_RecNoteDeal where RecNoteMain=(select ID from NM_RecNoteMain where docno='HHW-NM-250101003') and BusinessType=17))
delete from NM_NMToGL where DocID=(select ID from NM_RecNoteDeal where RecNoteMain=(select ID from NM_RecNoteMain where docno='HHW-NM-250101003') and BusinessType=17)
--删除行为核算信息
delete from NM_RecNoteAudit_Trl where ID=(select ID from NM_RecNoteAudit where RecNoteDeal=(select ID from NM_RecNoteDeal where id =(select ID from NM_RecNoteDeal where RecNoteMain=(select ID from NM_RecNoteMain where docno='HHW-NM-250101003') and BusinessType=17)))
delete from NM_RecNoteAudit where RecNoteDeal=(select ID from NM_RecNoteDeal where id =(select ID from NM_RecNoteDeal where RecNoteMain=(select ID from NM_RecNoteMain where docno='HHW-NM-250101003') and BusinessType=17))
--删除行为信息
delete from NM_RecNoteDeal_Trl where id =(select ID from NM_RecNoteDeal where RecNoteMain=(select ID from NM_RecNoteMain where docno='HHW-NM-250101003') and BusinessType=17)
delete from NM_RecNoteDeal where id =(select ID from NM_RecNoteDeal where RecNoteMain=(select ID from NM_RecNoteMain where docno='HHW-NM-250101003') and BusinessType=17)
--更新票据余额
update NM_RecNoteMain set Status=12,NoteBalanceFC=0,NoteBalanceMoney=0,LastNoteAction=(select ID from NM_RecNoteDeal where RecNoteMain=(select ID from NM_RecNoteMain where docno='HHW-NM-250101003') and BusinessType=11) where DocNo='HHW-NM-250101003'
