## 解决方案

declare @docno nvarchar(100)
declare @orgcode nvarchar(100)
set @docno ='单号2'
set @orgcode ='组织编码'
update l set l.Cancel_canceled=0,l.cancel_CancelReason=-1,l.cancel_CancelUser='',l.cancel_CancelDate=null
from pr_pr h
inner join pr_prline l on l.pr=h.id
inner join Base_Organization org on org.ID = h.Org
where h.docno =@docno and org.Code =@orgcode
update h set h.cancel_canceled=0,h.cancel_CancelReason=-1,h.cancel_CancelUser='',h.cancel_CancelDate=null
from pr_pr h
inner join pr_prline l on l.pr=h.id
inner join Base_Organization org on org.ID = h.Org
where h.docno =@docno and org.Code =@orgcode
