## 问题现象

系统报表无法满足客户需求，希望开发自定义报表

## 问题原因

系统无自定义报表模块，需在低代码平台开发相应应用并发布，作为应用功能安装到对应产品中

## 解决方案

开放平台编写脚本实现自定义报表。
1、开发文档地址：https://doc.chanapp.chanjet.com/；
2、存储过程如下：
create PROCEDURE kq_def_Report 
@createTime varchar (50), 
@saleOrderCode varchar (50),
@customerCode varchar (50),
@userid varchar (50) as BEGIN 
declare 
@tablename varchar (100) 
if len (@userid)<>0 
set @tablename = 'TEMP_0_kq_report' +'_' +@userid 
else 
set @tablename = 'TEMP_0_kq_report' 
exec( 'if exists(select name from sysobjects where id=object_id (N''' +@tablename +''' )
and OBJECTPROPERTY(id, N''IsUserTable'')= 1) drop table ' + @tablename) 
declare @startDate varchar(500) 
declare @endDate varchar(500) 
set @startDate = SUBSTRING(@createTime,1,CHARINDEX(',',@createTime)-1) 
set @endDate = SUBSTRING(@createTime,CHARINDEX(',',@createTime)+1,LEN(@createTime)) 
declare @sql varchar(3000)
set @sql='select 
a.code as xhdh
,c.code as jhdh
,e.code as khbm
,e.name as khmc
,b.pubuserdefdecm3 as xhdj
,d.pubuserdefdecm1 as jcbj
,b.taxPrice as xhsdj
,b.quantity as num
,b.taxRate as sl
,d.taxPrice as jhsdj
,(b.pubuserdefdecm3-d.pubuserdefdecm1) as cpml
,((b.taxPrice-b.pubuserdefdecm3)*b.quantity*b.taxRate)-(
(d.taxPrice-d.pubuserdefdecm1)*b.quantity*b.taxRate-(b.taxPrice-d.taxPrice)*0.18) as sjml
,(b.pubuserdefdecm3-d.pubuserdefdecm1)+
(((b.taxPrice-b.pubuserdefdecm3)*b.quantity*b.taxRate)-(
(d.taxPrice-d.pubuserdefdecm1)*b.quantity*b.taxRate-(b.taxPrice-d.taxPrice)*0.18)) as mlhs
into ' +@tablename + ' from SA_SaleDelivery a 
left join SA_SaleDelivery_b b on a.id=b.idSaleDeliveryDTO 
left join PU_PurchaseArrival c on b.receiveVoucherCode=c.code 
left join PU_PurchaseArrival_b d on c.id=d.idPurchaseArrivalDTO 
and b.idinventory=d.idinventory and b.batch = d.batch
left join AA_Partner e on a.idcustomer=e.id
where 1=1
and a.voucherdate >= '''+ @startDate +''' and a.voucherdate <=''' + @endDate +''''
if len (@saleOrderCode)<>0 
set @sql=@sql+'and a.code='''+@saleOrderCode+''''
if len (@customerCode)<>0 
set @sql=@sql+'and e.code='''+@customerCode+''''
--print @sql
exec(@sql) END
