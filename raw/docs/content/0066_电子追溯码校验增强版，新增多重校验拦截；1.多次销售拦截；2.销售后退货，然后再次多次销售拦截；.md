var ywlx = form.args.ds_djhz.field("ywlx").value;

if(ywlx == 'XS' ){
	var chk_imeino = DBUtil.uniqueValue("uf_sel_imeino",{"imeino":IMEINo});
	if(chk_imeino > 0){
		alert("追溯码已销售，不允许重复扫描！");
		sys_setFocus(textbox10);
		return;
	}
	
	var chk_amt = DBUtil.uniqueValue("uf_sel_ckamt",{"imeino":IMEINo});
	if(chk_amt>0){
		alert("追溯码已销售，不允许重复扫描！！");
		sys_setFocus(textbox10);
		return;
	}
}
if(ywlx == 'TH'){
	var chk_imeino = DBUtil.uniqueValue("uf_sel_imeino",{"imeino":IMEINo});

		alert("未查到追溯码的销售记录，不允许扫描录入！");
